Pengaturan & Script Google Sheet

Cara Deploy Google Apps Script:

  1. Buka spreadsheet Google Sheet Anda.
  2. Klik menu Ekstensi > Apps Script.
  3. Salin kode di bawah ini lalu tempel di editor script Google.
  4. Klik Deploy / Terapkan > Deployment Baru.
  5. Pilih Aplikasi Web. Atur Akses (Who has access) ke Siapa Saja (Anyone).
  6. Klik Terapkan dan salin URL Web App ke dalam kolom di atas.
Kode Apps Script (Code.gs): 
var SHEET_NAME = "Kegiatan harian";
var DROPDOWN_SHEET_NAME = "dropdown";

function doGet(e) {
  var action = e && e.parameter ? e.parameter.action : "";
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === "getDropdowns") {
    var katimList = [];
    var kegiatanList = [];
    var statusList = [];
    
    var dropdownSheet = ss.getSheetByName(DROPDOWN_SHEET_NAME);
    if (dropdownSheet) {
      var lastRow = dropdownSheet.getLastRow();
      if (lastRow >= 2) {
        var rangeValues = dropdownSheet.getRange("A2:C" + lastRow).getValues();
        rangeValues.forEach(function(row) {
          if (row[0] && row[0] !== "") katimList.push(row[0]);
          if (row[1] && row[1] !== "") kegiatanList.push(row[1]);
          if (row[2] && row[2] !== "") statusList.push(row[2]);
        });
      }
    }

    if (katimList.length === 0) katimList = ["Widya (Kasubag Tata Usaha)"];
    if (kegiatanList.length === 0) kegiatanList = ["Rapat"];
    if (statusList.length === 0) statusList = ["In Progress", "Completed"];

    return ContentService.createTextOutput(JSON.stringify({
      katim: katimList,
      kegiatan: kegiatanList,
      status: statusList
    })).setMimeType(ContentService.MimeType.JSON);
  }

  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) return ContentService.createTextOutput(JSON.stringify({ error: "Sheet tidak ditemukan" })).setMimeType(ContentService.MimeType.JSON);
  
  var data = sheet.getDataRange().getValues();
  var startRow = 0;
  for (var i = 0; i < data.length; i++) {
    if (data[i][1] === "No" || data[i][0] === "No") { startRow = i + 1; break; }
  }

  var result = [];
  for (var r = startRow; r < data.length; r++) {
    var row = data[r];
    if (!row[1] && !row[2]) continue;
    
    var tglRaw = row[7];
    var tglFormatted = "";
    if (tglRaw instanceof Date) {
      tglFormatted = Utilities.formatDate(tglRaw, Session.getScriptTimeZone(), "yyyy-MM-dd");
    } else if (typeof tglRaw === "string" && tglRaw.includes("/")) {
      var parts = tglRaw.split("/");
      if (parts.length === 3) tglFormatted = parts[2] + "-" + parts[1].padStart(2, '0') + "-" + parts[0].padStart(2, '0');
    } else {
      tglFormatted = String(tglRaw);
    }

    result.push({
      no: row[1],
      done: row[2] === true || String(row[2]).toUpperCase() === "TRUE",
      katim: row[3] || "",
      kegiatan: row[4] || "",
      rincian: row[5] || "",
      status: row[6] || "",
      tanggal: tglFormatted,
      sisaWaktu: String(row[8] || ""),
      dataDukung: row[9] || ""
    });
  }

  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sheet tidak ditemukan" })).setMimeType(ContentService.MimeType.JSON);
    }

    var contents = e.postData ? e.postData.contents : "";
    if (!contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Payload kosong" })).setMimeType(ContentService.MimeType.JSON);
    }

    var payload = JSON.parse(contents);
    var action = payload.action;
    var data = payload.data;

    function findRowByNo(targetNo) {
      var values = sheet.getDataRange().getValues();
      for (var i = 0; i < values.length; i++) {
        if (String(values[i][1]).trim() === String(targetNo).trim()) {
          return i + 1;
        }
      }
      return -1;
    }

    if (action === "add") {
      var parts = data.tanggal ? data.tanggal.split("-") : [];
      var formattedDate = parts.length === 3 ? parts[2] + "/" + parts[1] + "/" + parts[0] : data.tanggal;

      sheet.appendRow([
        "",
        data.no,
        data.done ? "TRUE" : "FALSE",
        data.katim || "",
        data.kegiatan || "",
        data.rincian || "",
        data.status || "",
        formattedDate || "",
        data.sisaWaktu || "",
        data.dataDukung || ""
      ]);

      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    } 
    else if (action === "edit") {
      var rowIndex = findRowByNo(data.no);
      if (rowIndex > 0) {
        var parts = data.tanggal ? data.tanggal.split("-") : [];
        var formattedDate = parts.length === 3 ? parts[2] + "/" + parts[1] + "/" + parts[0] : data.tanggal;

        sheet.getRange(rowIndex, 2).setValue(data.no);
        sheet.getRange(rowIndex, 3).setValue(data.done ? "TRUE" : "FALSE");
        sheet.getRange(rowIndex, 4).setValue(data.katim || "");
        sheet.getRange(rowIndex, 5).setValue(data.kegiatan || "");
        sheet.getRange(rowIndex, 6).setValue(data.rincian || "");
        sheet.getRange(rowIndex, 7).setValue(data.status || "");
        sheet.getRange(rowIndex, 8).setValue(formattedDate || "");
        sheet.getRange(rowIndex, 9).setValue(data.sisaWaktu || "");
        sheet.getRange(rowIndex, 10).setValue(data.dataDukung || "");

        return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Nomor kegiatan tidak ditemukan" })).setMimeType(ContentService.MimeType.JSON);
      }
    } 
    else if (action === "delete") {
      var rowIndex = findRowByNo(data.no);
      if (rowIndex > 0) {
        sheet.deleteRow(rowIndex);
        return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Nomor kegiatan untuk dihapus tidak ditemukan" })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Aksi tidak dikenali" })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
